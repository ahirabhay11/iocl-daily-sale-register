const DB_FILE_NAME = "IOCL_Daily_Sale_Register_DB.json";

function getDbFile() {
  const files = DriveApp.getFilesByName(DB_FILE_NAME);
  if (files.hasNext()) return files.next();
  return DriveApp.createFile(DB_FILE_NAME, JSON.stringify({ archive: {}, updatedAt: Date.now() }), MimeType.PLAIN_TEXT);
}

function doGet(e) {
  const file = getDbFile();
  const content = file.getBlob().getDataAsString() || '{"archive":{}}';
  return ContentService.createTextOutput(content).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents || "{}");
  const file = getDbFile();
  let current = {};
  try { current = JSON.parse(file.getBlob().getDataAsString() || "{}"); } catch (_) {}
  const mergedArchive = Object.assign({}, current.archive || {}, body.archive || {});
  const payload = { updatedAt: Date.now(), archive: mergedArchive };
  file.setContent(JSON.stringify(payload));
  return ContentService.createTextOutput(JSON.stringify({ ok: true, dates: Object.keys(mergedArchive).length }))
    .setMimeType(ContentService.MimeType.JSON);
}
